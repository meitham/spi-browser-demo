The browser handles HTTP and HTTPS. The WASM module handles SPI objects. This example requires no libcurl or application backend.

const response = await fetch(url);
if (!response.ok) throw new Error(response.status);

const object = spi.Object.fromString(
  await response.text()
);
try {
  console.log(object.toString('TEXT'));
} finally {
  object.delete();
}
